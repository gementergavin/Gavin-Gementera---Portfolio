! =====================================================================
! R1 - Pacific Crest USD edge router (Cisco ISR4331 in Packet Tracer)
! Router-on-a-stick for VLANs 10/20/30/40/99, DHCP, ACLs, NAT to the ISP.
!
! Paste into the R1 CLI from the "Router>" prompt.
! Passwords are lab placeholders: change CHANGE_ME_* before you use them.
! =====================================================================
enable
configure terminal
hostname R1
no ip domain-lookup
ip domain-name pcusd.internal
enable secret CHANGE_ME_ENABLE
username netadmin secret CHANGE_ME_ADMIN
service password-encryption
banner motd # Authorized access only - Pacific Crest USD network #
!
! ---------- DHCP: first 20 addresses in each VLAN are reserved ----------
ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.30.1 10.10.30.20
!
ip dhcp pool STAFF
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.40.10
 domain-name pcusd.internal
ip dhcp pool STUDENTS
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.40.10
 domain-name pcusd.internal
ip dhcp pool GUEST
 network 10.10.30.0 255.255.255.0
 default-router 10.10.30.1
 dns-server 198.51.100.10
!
! ---------- ACLs (applied inbound on the student and guest VLANs) ----------
ip access-list extended STUDENTS-IN
 remark Students may answer staff (replies only) but not start connections to staff
 permit tcp 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255 established
 permit icmp 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255 echo-reply
 deny ip 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255
 remark Students may not reach network device management
 deny ip 10.10.20.0 0.0.0.255 10.10.99.0 0.0.0.255
 remark Everything else (servers, internet) is allowed
 permit ip any any
!
ip access-list extended GUEST-IN
 remark Let new guest devices ask for a DHCP address
 permit udp any any eq bootps
 remark Guests may not reach anything inside the district (10.0.0.0/8)
 deny ip 10.10.30.0 0.0.0.255 10.0.0.0 0.255.255.255
 remark Internet only
 permit ip 10.10.30.0 0.0.0.255 any
!
! ---------- Interfaces ----------
interface GigabitEthernet0/0/0
 description Uplink to ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
!
interface GigabitEthernet0/0/1
 description Trunk to SW-CORE Gi0/1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0/1.10
 description VLAN 10 STAFF
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/0/1.20
 description VLAN 20 STUDENTS
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.0
 ip access-group STUDENTS-IN in
 ip nat inside
!
interface GigabitEthernet0/0/1.30
 description VLAN 30 GUEST
 encapsulation dot1Q 30
 ip address 10.10.30.1 255.255.255.0
 ip access-group GUEST-IN in
 ip nat inside
!
interface GigabitEthernet0/0/1.40
 description VLAN 40 SERVERS
 encapsulation dot1Q 40
 ip address 10.10.40.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/0/1.99
 description VLAN 99 MANAGEMENT
 encapsulation dot1Q 99
 ip address 10.10.99.1 255.255.255.0
!
! ---------- NAT: all district VLANs share the one public address ----------
access-list 1 remark District inside networks for NAT
access-list 1 permit 10.10.0.0 0.0.255.255
ip nat inside source list 1 interface GigabitEthernet0/0/0 overload
!
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! ---------- Remote management: SSH only, local accounts ----------
line console 0
 logging synchronous
 login local
line vty 0 4
 login local
 transport input ssh
!
end
write memory
