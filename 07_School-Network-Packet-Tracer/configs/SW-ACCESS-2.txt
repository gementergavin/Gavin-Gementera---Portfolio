! =====================================================================
! SW-ACCESS-2 - classroom wing access switch (Cisco 2960-24TT)
! Student computers (VLAN 20) and teacher desks (VLAN 10).
! Paste from the "Switch>" prompt. Change CHANGE_ME_* passwords.
! =====================================================================
enable
configure terminal
hostname SW-ACCESS-2
no ip domain-lookup
ip domain-name pcusd.internal
enable secret CHANGE_ME_ENABLE
username netadmin secret CHANGE_ME_ADMIN
service password-encryption
banner motd # Authorized access only - Pacific Crest USD network #
!
vtp mode transparent
!
vlan 10
 name STAFF
vlan 20
 name STUDENTS
vlan 30
 name GUEST
vlan 40
 name SERVERS
vlan 99
 name MANAGEMENT
vlan 999
 name BLACKHOLE
vlan 1000
 name NATIVE
!
! ---------- Uplink ----------
interface FastEthernet0/24
 description Trunk to SW-CORE Fa0/24
 switchport mode trunk
 switchport trunk native vlan 1000
 switchport trunk allowed vlan 10,20,30,40,99,1000
 switchport nonegotiate
!
! ---------- Student computers ----------
interface range FastEthernet0/1 - 16
 description Student computer
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
!
! ---------- Teacher desks ----------
interface range FastEthernet0/17 - 18
 description Teacher desk
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
!
! ---------- Unused ----------
interface range FastEthernet0/19 - 23
 description Unused
 switchport mode access
 switchport access vlan 999
 shutdown
!
interface range GigabitEthernet0/1 - 2
 description Unused
 switchport mode access
 switchport access vlan 999
 shutdown
!
! ---------- Management ----------
interface Vlan1
 no ip address
 shutdown
!
interface Vlan99
 description Management
 ip address 10.10.99.4 255.255.255.0
 no shutdown
!
ip default-gateway 10.10.99.1
!
line console 0
 logging synchronous
 login local
line vty 0 15
 login local
 transport input ssh
!
end
write memory
